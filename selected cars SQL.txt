-- create main table cars with 10 columns and various Oracle data types
CREATE TABLE cars (
    car_id          INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    vin_number      VARCHAR2(17),
    make            VARCHAR2(50),
    model           VARCHAR2(50),
    model_year      INTEGER,
    vehicle_type    VARCHAR2(30),
    mileage         INTEGER,
    price           NUMBER(8,2),
    date_acquired   DATE,
);
